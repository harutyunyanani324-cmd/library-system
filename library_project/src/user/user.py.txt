class User:
    def __init__(self, name):
        self.name = name
        self.borrowed_items = []

    def borrow_item(self, item):
        if item.borrow():
            self.borrowed_items.append(item)
            print(f"User {self.name} borrowed: {item.title}")
        else:
            print(f"Item {item.title} is already borrowed.")

    def return_item(self, item):
        if item in self.borrowed_items:
            item.return_item()
            self.borrowed_items.remove(item)
            print(f"User {self.name} returned: {item.title}")

    def list_borrowed_items(self):
        titles = [item.title for item in self.borrowed_items]
        print(f"{self.name}'s borrowed items: {titles}")
