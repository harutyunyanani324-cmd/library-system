class Library:
    def __init__(self):
        self.items = []

    def add_item(self, item):
        self.items.append(item)

    def display_all_items(self):
        for item in self.items:
            item.display_info()

    def __len__(self):
        return len(self.items)

    def __str__(self):
        return f"Library contains {len(self.items)} items"
